需要貼入 Google Apps Script 的 doPost 後端程式：

請至試算表的 Apps Script 專案中，新增並發布為「網頁應用程式」：

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var code = (data.code || "").trim();
    var ss = SpreadsheetApp.openById("1zJkGLtCQC1ZZPbQGJiQkxSWec2wRtlKA1pGfmoHaDEY");
    var sheet = ss.getSheetByName("Form_Responses") || ss.getSheets()[0];
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return ContentService.createTextOutput(JSON.stringify({status: 'not_found'}));
    
    // 取得 E(編號), F(狀態), G(時間)
    var codes = sheet.getRange(2, 5, lastRow - 1, 1).getValues().flat();
    var targetRow = -1;
    for (var i = 0; i < codes.length; i++) {
      if (String(codes[i]).trim() === code) {
        targetRow = i + 2;
        break;
      }
    }
    
    if (targetRow === -1) {
      return ContentService.createTextOutput(JSON.stringify({status: 'not_found'})).setMimeType(ContentService.MimeType.JSON);
    }
    
    var name = sheet.getRange(targetRow, 2).getValue();
    var phone = sheet.getRange(targetRow, 3).getValue();
    var email = sheet.getRange(targetRow, 4).getValue();
    var currentStatus = sheet.getRange(targetRow, 6).getValue();
    var currentTime = sheet.getRange(targetRow, 7).getValue();
    
    if (currentStatus === "已報到") {
      return ContentService.createTextOutput(JSON.stringify({
        status: 'already_checked',
        name: name,
        phone: phone,
        email: email,
        code: code,
        checkinTime: currentTime
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var nowStr = Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy/MM/dd HH:mm:ss");
    sheet.getRange(targetRow, 6).setValue("已報到");
    sheet.getRange(targetRow, 7).setValue(nowStr);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      name: name,
      phone: phone,
      email: email,
      code: code,
      checkinTime: nowStr
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
